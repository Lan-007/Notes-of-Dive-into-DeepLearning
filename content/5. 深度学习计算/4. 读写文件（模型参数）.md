### 2. 淇濆瓨鍙傛暟

```python
from torch import nn

class MLP(nn.Module):
    def __init__(self):
        super().__init__()

        self.hidden = nn.Linear(20, 256)
        self.output = nn.Linear(256, 10)

    def forward(self, x):
        x = torch.relu(self.hidden(x))
        return self.output(x)

net = MLP()

# 淇濆瓨鍙傛暟
torch.save(net.state_dict(), 'mlp.params')  # 淇濆瓨妯″瀷鍙傛暟瀛楀吀

# 鍔犺浇鍙傛暟
clone = MLP()

clone.load_state_dict(torch.load('mlp.params')) # 璇诲彇鍙傛暟

clone.eval() # 鍒囨崲鍒拌瘎浼版ā寮?```

def knn(W, x, k):
    cos = torch.mv(W, x.reshape(-1,)) / (
        torch.sqrt(torch.sum(W * W, axis=1) + 1e-9)
        * torch.sqrt((x * x).sum())
    )
    _, topk = torch.topk(cos, k=k)              # 鎵惧嚭鐩镐技搴︽渶楂樼殑 k 涓瘝
    return topk, [cos[int(i)] for i in topk]

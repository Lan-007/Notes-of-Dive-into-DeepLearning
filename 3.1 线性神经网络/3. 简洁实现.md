```python
num_epochs = 3
data_iter = load_array((features, labels), batch_size=10)

for epoch in range(num_epochs):
	for X, y in data_iter:
		l = loss (net(X), y)
		
		trainer.zero_grad() # 清除上一批梯度
		l.bvackward()       # 反向传播 
		trainer.step()      # 更新梯度
		
	l = loss(net(features), labels)
	print(f'epoch {epoch+1}, loss {l:f}')
```

```python
import torch
def batchify(data):
    # 瀹氫箟涓€涓嚱鏁帮紝鎶婂涓牱鏈暣鐞嗘垚涓€涓?batch
    # data 鏄竴涓垪琛紝閲岄潰姣忎釜鍏冪礌鏄竴涓缁冩牱鏈?    # 姣忎釜鏍锋湰褰㈠锛?    # (center, context, negative)

    max_len = max(len(context) + len(negative) for _, context, negative in data)
    # 鎵惧嚭鏈€闀跨殑 鈥滀笂涓嬫枃璇?+ 璐熸牱鏈瘝鈥?闀垮害锛岀敤浜巔adding

    centers = []
    contexts_negatives = []
    masks = []
    labels = []

    for center, context, negative in data:

        cur_len = len(context) + len(negative)

        centers.append(center)

        contexts_negatives.append(
            context + negative + [0] * (max_len - cur_len)
        )

        masks.append(
            [1] * cur_len + [0] * (max_len - cur_len)
        )

        labels.append(
            [1] * len(context) + [0] * (max_len - len(context))
        )

    return (
        torch.tensor(centers).reshape((-1, 1)),
        # 褰㈢姸鏄?[batch_size, 1]

        torch.tensor(contexts_negatives),
        # 褰㈢姸鏄?[batch_size, max_len]
        torch.tensor(masks),
        # 褰㈢姸鏄?[batch_size, max_len]
        torch.tensor(labels)
        # 褰㈢姸鏄?[batch_size, max_len]
    )
```

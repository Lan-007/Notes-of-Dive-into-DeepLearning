# 参数管理

模型搭好之后，如何管理参数。

对于一个简单的模型MLP：

```python
import torch
from torch import nn

net = nn.Sequential(nn.Linear(4, 8),
					nn.ReLU(),
					nn.Linear(8,1))
X = torch.rand(size=(2,4))
```

### 1. 访问层和参数

```python
net[0] # 访问第 0 层 nn.Linear(4, 8)
net[1] # 访问第 1 层 nn.ReLU()

print(net[2])
# Linear(in_features=8, out_features=1, bias=True)

net[2].state_dict()
#访问第 2 层的 参数字典:key 是参数名，value 是参数张量。
# OrderedDict([
#    ('weight', tensor([[...]])),
#    ('bias', tensor([...]))
# ])

type(net[2].bias)  # torch.nn.parameter.Parameter
net[2].weight      # 查看最后一层权重，是 Parameter 类型
net[2].weight.data # 查看最后一层权重里的具体数值
net[2].weight.grad # 查看最后一层权重的梯度
```

一次性访问/遍历所有参数

```python
for name, param in net.named_parameters():  # 遍历所有参数，同时拿到参数名和参数值
    print(name, param.shape)                # 打印参数名和参数形状
# 0.weight torch.Size([8, 4])
# 0.bias torch.Size([8])
# 2.weight torch.Size([1, 8])
# 2.bias torch.Size([1])
```

嵌套块中的参数访问

```python
net = nn.Sequential(
    nn.Sequential(
        nn.Linear(4, 8),
        nn.ReLU()
    ),
    nn.Linear(8, 1)
)

net[0][0] # Linear(4, 8)
net[0][0].weight
```

### 2. 参数初始化

把权重初始化为正态分布

```python
def init_normal(m):                         # 初始化函数
    if type(m) == nn.Linear:
        nn.init.normal_(m.weight, mean=0, std=0.01)  # 正态分布
        # nn.init.constant_(m.weight, 1) 初始化为常数
        nn.init.zeros_(m.bias)              # 把偏置初始化为 0

net.apply(init_normal)                      # 对每一层应用初始化函数
```

后面带 `_`，表示原地修改。

Xavier初始化：

```python
def init_xavier(m):                         # Xavier初始化函数
    if type(m) == nn.Linear:                # 如果当前层是 Linear
        nn.init.xavier_uniform_(m.weight)   # 用 Xavier 均匀分布初始化权重

net.apply(init_xavier)
net[0].apply(init_xavier) # 对第 0 层应用 Xavier 初始化

net[0].weight.data[:] += 1 # 把第 0 层所有权重都加 1
```

## 3. 参数绑定（Parameter Sharing）

```python
shared = nn.Linear(8, 8)  # 创建一个共享的 Linear 层，输入 8 维，输出 8 维

net = nn.Sequential(
    nn.Linear(4, 8),
    nn.ReLU(),
    shared,               # 第 2 层：使用 shared 这个 Linear
    nn.ReLU(),
    shared,               # 第 4 层：再次使用同一个 shared Linear
    nn.ReLU(),
    nn.Linear(8, 1)
)
```

## 4. 保存和加载参数

```python
# 保存参数
torch.save(
    net.state_dict(),
    "model.params"
)

# 加载参数
net2 = nn.Sequential(
    nn.Linear(4,8),
    nn.ReLU(),
    nn.Linear(8,1)
)

net2.load_state_dict(
    torch.load("model.params")
)
```
